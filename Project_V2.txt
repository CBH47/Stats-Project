Some example code:

y_vector <- rnorm(34, mean = 4, sd = 2.1)
x_vector <- y_vector + rnorm(34, mean = 2, 4) # Add some noise to our X predictors
sample_of_15 <- data.frame(y_vec = y_vector, x_vec = x_vector)
correlation_coefs <- numeric(10) # preallocating space
for (i in 1:10) {
 bootstrap_sampled_idxs <- sample(1:15, 15, replace = TRUE)
 new_df <- sample_of_15[bootstrap_sampled_idxs, ]
 linear_model_res <- lm(new_df$y_vec ~ new_df$x_vec)
 correlation_coefs[i] <- sqrt(summary(linear_model_res)$r.squared)
}
